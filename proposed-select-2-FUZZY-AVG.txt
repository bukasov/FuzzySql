-- SQL:

SELECT lvv linguistic_variable_value, ROUND(SUM(weight * value) / SUM(weight), 2) AS FUZZY_AVG
FROM
(SELECT entity_id, entity_name, lvv, y1+(NUMBER_VALUE-x1)*(y2-y1)/(x2-x1) AS weight
 FROM entity NATURAL JOIN attribute NATURAL JOIN value NATURAL JOIN lvv NATURAL JOIN mf
 WHERE NUMBER_VALUE >= x1 AND NUMBER_VALUE < x2 AND attribute_id=2 /*year*/)
JOIN
(SELECT entity_id, number_value AS value
 FROM entity NATURAL JOIN attribute NATURAL JOIN value
 WHERE attribute_id=4 /*price*/)
USING (entity_id)
GROUP BY lvv;
