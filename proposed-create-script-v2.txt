DROP TABLE mf;
DROP TABLE lvv;
DROP TABLE value;
DROP TABLE entity;
DROP TABLE attribute;
DROP TABLE class;

CREATE TABLE class (
    class_id NUMBER PRIMARY KEY,
    class_name VARCHAR2(50)
);
-- add index on 'name' for fast search by name if needed

CREATE TABLE attribute (    
    attribute_id NUMBER PRIMARY KEY,
    class_id NUMBER,
    attribute_name VARCHAR2(50),
    attribute_type VARCHAR2(6) -- 'TEXT' / 'NUMBER'
);

CREATE TABLE entity (    
    entity_id NUMBER PRIMARY KEY,
    class_id NUMBER,
    entity_name VARCHAR2(50)
);
-- add index on 'name' for fast search by name if needed

CREATE TABLE value (
    value_id NUMBER PRIMARY KEY,
    entity_id NUMBER,
    attribute_id NUMBER,
    text_value VARCHAR2(2000), -- if attribute.type = 'TEXT'
    number_value NUMBER        -- if attribute.type = 'NUMBER'
);

ALTER TABLE entity ADD CONSTRAINT entity_class_fk FOREIGN KEY(class_id) REFERENCES class;
ALTER TABLE attribute ADD CONSTRAINT attribute_class_fk FOREIGN KEY(class_id) REFERENCES class;
ALTER TABLE value ADD CONSTRAINT value_entity_fk FOREIGN KEY(entity_id) REFERENCES entity;
ALTER TABLE value ADD CONSTRAINT value_attribute_fk FOREIGN KEY(attribute_id) REFERENCES attribute;
ALTER TABLE value ADD CONSTRAINT value_entity_attribute_uq UNIQUE (entity_id, attribute_id);

CREATE INDEX attribute_class_id_idx ON attribute (class_id);
CREATE INDEX value_entity_id_idx ON value (entity_id);
CREATE INDEX value_attribute_id_idx ON value (attribute_id);
CREATE INDEX value_number_value_idx ON value (number_value);


CREATE TABLE lvv (
    lvv_id NUMBER PRIMARY KEY,
    attribute_id NUMBER,
    lvv VARCHAR2(50)
);
ALTER TABLE lvv ADD CONSTRAINT lvv_attribute_fk FOREIGN KEY(attribute_id) REFERENCES attribute;
CREATE INDEX lvv_attribute_id_idx ON lvv (attribute_id);
CREATE INDEX lvv_lvv_idx ON lvv (lvv);

CREATE TABLE mf (
    mf_id NUMBER PRIMARY KEY,
    lvv_id NUMBER,
    x1 NUMBER, -- inclusive
    y1 NUMBER, -- 0..1
    x2 NUMBER, -- exclusive
    y2 NUMBER  -- 0..1
);
ALTER TABLE mf ADD CONSTRAINT mf_lvv_fk FOREIGN KEY(lvv_id) REFERENCES lvv;
ALTER TABLE mf ADD CONSTRAINT mf_x_order_chk CHECK (x2 > x1);
ALTER TABLE mf ADD CONSTRAINT mf_y1_chk CHECK (y1 BETWEEN 0 AND 1);
ALTER TABLE mf ADD CONSTRAINT mf_y2_chk CHECK (y2 BETWEEN 0 AND 1);
CREATE INDEX mf_lvv_id_idx ON mf (lvv_id);
