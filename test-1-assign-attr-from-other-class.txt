SELECT entity_id, attribute_id
FROM value 
    JOIN entity USING(entity_id) JOIN class c1 ON entity.class_id = c1.class_id
    JOIN attribute USING(attribute_id) JOIN class c2 ON attribute.class_id = c2.class_id
WHERE c1.class_id <> c2.class_id;
