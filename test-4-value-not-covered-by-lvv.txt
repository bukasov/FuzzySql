WITH 
lvv_ranges AS (
  SELECT attribute_id, lvv_id, min(x1) AS left, max(x2) AS right
  FROM mf JOIN lvv USING(lvv_id)
  GROUP BY attribute_id, lvv_id ORDER BY lvv_id),
lvv_ranges_without_subranges AS (  
  SELECT * FROM lvv_ranges WHERE lvv_id NOT IN (
    SELECT a.lvv_id
    FROM lvv_ranges a JOIN lvv_ranges b ON a.attribute_id = b.attribute_id
    WHERE a.left >= b.left AND a.right <=b.right AND a.lvv_id<>b.lvv_id)
),
current_right_with_previous_left AS (
    SELECT attribute_id, lvv_id, left AS current_left, LAG(right) OVER (PARTITION BY attribute_id ORDER BY left) AS previous_right
    FROM lvv_ranges_without_subranges)
SELECT attribute_id, lvv_id, current_left, previous_right
FROM current_right_with_previous_left
WHERE previous_right < current_left;
